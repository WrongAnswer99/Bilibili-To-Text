cmake_minimum_required(VERSION 3.16)
project(BilibiliToText_whisper LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(WHISPER_BACKEND "cuda" CACHE STRING "Local Whisper backend: cuda or cpu")
set_property(CACHE WHISPER_BACKEND PROPERTY STRINGS cuda cpu)
set(WHISPER_SOURCE "${CMAKE_SOURCE_DIR}/third_party/whisper.cpp")
if(NOT EXISTS "${WHISPER_SOURCE}/include/whisper.h")
  message(FATAL_ERROR "Missing project-local Whisper source; run tools/setup_whisper.ps1")
endif()
add_library(whisper_bridge SHARED native/asr_c_api.cpp native/audio.cpp)
target_include_directories(whisper_bridge PRIVATE native)
target_compile_definitions(whisper_bridge PRIVATE ASR_BUILD_DLL NOMINMAX ASR_WHISPER_VERSION="1.9.5")
target_link_libraries(whisper_bridge PRIVATE mfplat mfreadwrite mfuuid ole32)
set_target_properties(whisper_bridge PROPERTIES PREFIX "" OUTPUT_NAME "whisper_bridge"
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/runtime"
  RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/runtime")
if(WHISPER_BACKEND STREQUAL "cuda")
  file(SHA256 "${WHISPER_SOURCE}/include/whisper.h" WHISPER_HEADER_SHA)
  if(NOT WHISPER_HEADER_SHA STREQUAL "a7d19f7feb5be52426628ff07e0602de28a30dc4312d0d0603e1e536753f76dd")
    message(FATAL_ERROR "Whisper header ABI does not match pinned b5454 runtime")
  endif()
  foreach(dll whisper ggml ggml-base ggml-cuda cublas64_12 cublasLt64_12 cudart64_12)
    if(NOT EXISTS "${CMAKE_SOURCE_DIR}/runtime/${dll}.dll")
      message(FATAL_ERROR "Missing local ${dll}.dll; run tools/setup_whisper_gpu.ps1")
    endif()
  endforeach()
  target_include_directories(whisper_bridge PRIVATE "${WHISPER_SOURCE}/include" "${WHISPER_SOURCE}/ggml/include")
  target_compile_definitions(whisper_bridge PRIVATE ASR_CUDA_RUNTIME)
elseif(WHISPER_BACKEND STREQUAL "cpu")
  enable_language(C)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
  set(WHISPER_BUILD_IS_DEV OFF CACHE BOOL "" FORCE)
  set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
  set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
  set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
  set(GGML_CUDA OFF CACHE BOOL "" FORCE)
  set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
  add_subdirectory("${WHISPER_SOURCE}" vendor-whisper EXCLUDE_FROM_ALL)
  target_link_libraries(whisper_bridge PRIVATE whisper)
else()
  message(FATAL_ERROR "WHISPER_BACKEND must be cuda or cpu")
endif()
if(MSVC)
  target_compile_options(whisper_bridge PRIVATE /O2 /utf-8)
elseif(MINGW)
  target_compile_options(whisper_bridge PRIVATE -O2)
  target_link_options(whisper_bridge PRIVATE -static -static-libgcc -static-libstdc++)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(native_utf8_test tests/native_utf8_test.cpp)
  target_include_directories(native_utf8_test PRIVATE native)
  if(MSVC)
    target_compile_options(native_utf8_test PRIVATE /UNDEBUG)
  else()
    target_compile_options(native_utf8_test PRIVATE -UNDEBUG)
  endif()
  add_test(NAME native_utf8 COMMAND native_utf8_test)
endif()
